function SlideView({ slide }: { slide: Slide }) { return (<div className="slide-grid"><article className="content-zone"><p className="eyebrow">{slide.eyebrow}</p><h1>{slide.title}</h1>{slide.subtitle && <p className="subtitle">{slide.subtitle}</p>}{slide.body?.map((paragraph) => <p className="body-text" key={paragraph}>{paragraph}</p>)}{slide.bullets && <BulletList items={slide.bullets} />}{slide.flow && <Flow items={slide.flow} />}{slide.quote && <blockquote>{slide.quote}</blockquote>}</article><aside className="visual-zone">{slide.cards ? <Cards cards={slide.cards} /> : <AnimatedVisual type={slide.type} />}</aside></div>); }
function BulletList({ items }: { items: string[] }) { return <ul className="bullet-list">{items.map((item,index)=><li style={{animationDelay:`${index*70}ms`}} key={item}>{item}</li>)}</ul>; }
function Cards({ cards }: { cards: Card[] }) { return <div className="cards">{cards.map((card,index)=><article className={`card tone-${card.tone ?? "blue"}`} style={{animationDelay:`${index*80}ms`}} key={card.title+card.text}><h2>{card.title}</h2><p>{card.text}</p></article>)}</div>; }
function Flow({ items }: { items: string[] }) { return <div className="flow">{items.map((item,index)=><div className="flow-item" style={{animationDelay:`${index*95}ms`}} key={item}><span>{item}</span>{index<items.length-1&&<b>→</b>}</div>)}</div>; }
