      <div className="progress"><div style={{ width: `${progress}%` }} /></div>

      <div className="stage">
        <AnimatePresence mode="wait">
          <motion.section
            key={current}
            className="slide"
            initial={{ opacity: 0, y: 20, scale: 0.99 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={{ opacity: 0, y: -20, scale: 0.99 }}
            transition={{ duration: 0.35, ease: "easeOut" }}
          >
            <div className="slide-grid">
              <article className="content-zone">
                <motion.p className="eyebrow" initial={{ opacity: 0, x: -16 }} animate={{ opacity: 1, x: 0 }} transition={{ delay: 0.05 }}>
                  {slide.eyebrow}
                </motion.p>
                <motion.h1 initial={{ opacity: 0, x: -18 }} animate={{ opacity: 1, x: 0 }} transition={{ delay: 0.1 }}>
                  {slide.title}
                </motion.h1>
                {slide.subtitle && (
                  <motion.p className="subtitle" initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.15 }}>
                    {slide.subtitle}
                  </motion.p>
                )}
                {slide.body?.map((paragraph, index) => (
                  <motion.p className="body-text" key={paragraph} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.18 + index * 0.04 }}>
                    {paragraph}
                  </motion.p>
                ))}
                {slide.bullets && (
                  <ul className="bullet-list">
                    {slide.bullets.map((item, index) => (
                      <motion.li key={item} initial={{ opacity: 0, x: -10 }} animate={{ opacity: 1, x: 0 }} transition={{ delay: 0.2 + index * 0.035 }}>
                        {item}
                      </motion.li>
                    ))}
                  </ul>
                )}
                {slide.flow && (
                  <div className="flow">
                    {slide.flow.map((item, index) => (
                      <motion.div className="flow-item" key={item} initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} transition={{ delay: 0.18 + index * 0.05 }}>
                        <span>{item}</span>
                        {index < slide.flow!.length - 1 && <b>→</b>}
                      </motion.div>
                    ))}
                  </div>
                )}
                {slide.code && (
                  <motion.pre className="code-block" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.22 }}>
                    <code>{slide.code}</code>
                  </motion.pre>
                )}
                {slide.note && (
                  <motion.blockquote initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.24 }}>
                    {slide.note}
                  </motion.blockquote>
                )}
              </article>

              <aside className="visual-zone">
                {slide.cards ? (
                  <div className="cards">
                    {slide.cards.map((card, index) => (
                      <motion.article
                        key={card.title + card.text}
                        className={`card tone-${card.tone ?? "wine"}`}
                        initial={{ opacity: 0, y: 16 }}
                        animate={{ opacity: 1, y: 0 }}
                        transition={{ delay: 0.15 + index * 0.06 }}
                      >
                        <h2>{card.title}</h2>
                        <p>{card.text}</p>
                      </motion.article>
                    ))}
                  </div>
                ) : (
                  <Visual label={slide.visualLabel ?? "TEST"} />
                )}
              </aside>
            </div>
          </motion.section>
        </AnimatePresence>
      </div>

      <footer className="deck-footer">
        <span>← → navegar</span>
        <span>F pantalla completa</span>
        <span>Home / End</span>
      </footer>
    </main>
  );
}
