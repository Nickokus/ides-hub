  return <motion.button disabled={disabled} whileHover={!disabled ? { y: -2, scale: 1.015 } : {}} whileTap={!disabled ? { scale: .965 } : {}} onClick={onClick} className={`btn ${kind} ${active ? 'active' : ''} ${className}`}>{children}</motion.button>
