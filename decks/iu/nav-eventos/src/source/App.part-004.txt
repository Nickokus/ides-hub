function FeedbackDemo(){
  const [status,setStatus]=useState<'idle'|'loading'|'success'|'error'>('idle')
  const run=(ok:boolean)=>{setStatus('loading'); setTimeout(()=>setStatus(ok?'success':'error'),900)}
  return <div className="feedbackDemo">
    <div className={`feedbackScreen ${status}`}>
      <AnimatePresence mode="wait"><motion.div key={status} initial={{opacity:0,y:15}} animate={{opacity:1,y:0}} exit={{opacity:0,y:-15}}>
        {status==='idle' && <><h3>Formulario listo</h3><p>Esperando una acción del usuario.</p></>}
        {status==='loading' && <><div className="spinner"/><h3>Guardando…</h3><p>El sistema confirma que recibió el evento.</p></>}
        {status==='success' && <><h3>✓ Guardado correctamente</h3><p>El usuario sabe que la acción terminó bien.</p></>}
        {status==='error' && <><h3>⚠ No pudimos guardar</h3><p>El feedback explica el problema y permite intentar de nuevo.</p></>}
      </motion.div></AnimatePresence>
    </div>
    <div className="segmented"><Btn onClick={()=>run(true)}>Simular éxito</Btn><Btn kind="danger" onClick={()=>run(false)}>Simular error</Btn><Btn kind="secondary" onClick={()=>setStatus('idle')}>Volver al inicio</Btn></div>
    <p className="caption teacherCaption">Probamos primero éxito y después error. En ambos casos quiero que vean que el feedback elimina la duda: el usuario sabe si el sistema sigue trabajando, terminó bien o necesita otra acción.</p>
  </div>
}
