export default function App() {
  const [index, setIndex] = useState(0)
  const [dir, setDir] = useState(1)
  const [fullscreen, setFullscreen] = useState(false)
  const slide = slides[index]
  const progress = useMemo(() => ((index + 1) / slides.length) * 100, [index])

  const go = (i: number) => {
    const next = Math.max(0, Math.min(slides.length - 1, i))
    setDir(next >= index ? 1 : -1)
    setIndex(next)
  }
  const next = () => go(index + 1)
  const prev = () => go(index - 1)
  const toggleFullscreen = async () => {
    if (!document.fullscreenElement) await document.documentElement.requestFullscreen()
    else await document.exitFullscreen()
  }

  useEffect(() => {
    const sync = () => setFullscreen(Boolean(document.fullscreenElement))
    document.addEventListener('fullscreenchange', sync)
    return () => document.removeEventListener('fullscreenchange', sync)
  }, [])
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      const target = e.target as HTMLElement | null
      const isInteractive = Boolean(target?.closest('button, input, textarea, select, .react-flow'))
      if (isInteractive && e.key.toLowerCase() !== 'f') return
      if (['ArrowRight', 'PageDown', ' '].includes(e.key)) { e.preventDefault(); next() }
      if (['ArrowLeft', 'PageUp'].includes(e.key)) { e.preventDefault(); prev() }
      if (e.key === 'Home') go(0)
      if (e.key === 'End') go(slides.length - 1)
      if (e.key.toLowerCase() === 'f') toggleFullscreen()
    }
    window.addEventListener('keydown', handler)
    return () => window.removeEventListener('keydown', handler)
  }, [index])

  return (
    <main className="presentation">
      <div className="ambient amb-a" />
      <div className="ambient amb-b" />
      <header className="topbar">
        <div className="brand"><div className="brand-mark"><Workflow size={22} /></div><div><strong>Interfaz de Usuario</strong><span>User Flow profesional · presentación interactiva</span></div></div>
        <div className="top-actions"><span>{String(index + 1).padStart(2, '0')} / {slides.length}</span><button onClick={toggleFullscreen}>{fullscreen ? <Minimize2 size={18} /> : <Expand size={18} />} F</button></div>
      </header>
      <div className="progress-track"><motion.div animate={{ width: `${progress}%` }} transition={{ ease, duration: .38 }} /></div>

      <section className="stage">
        <AnimatePresence mode="wait" custom={dir}>
          {slide.type === 'flow-interactive' ? (
            <motion.article key={index} custom={dir} variants={{ enter: (d: number) => ({ opacity: 0, x: d > 0 ? 90 : -90, scale: .985 }), center: { opacity: 1, x: 0, scale: 1 }, exit: (d: number) => ({ opacity: 0, x: d > 0 ? -70 : 70, scale: .985 }) }} initial="enter" animate="center" exit="exit" transition={{ duration: .42, ease }} className="slide flow-slide">
              <div className="flow-slide-heading"><div><div className="kicker">{slide.kicker}</div><h2>{slide.title}</h2><p>{slide.subtitle}</p></div><div className="flow-slide-tip">Hacé click en cada nodo para explicar su función y seguí las flechas para recorrer el flujo.</div></div>
              <div className="flow-full-visual"><InteractiveFlowVisual /></div>
            </motion.article>
          ) : (
            <motion.article key={index} custom={dir} variants={{ enter: (d: number) => ({ opacity: 0, x: d > 0 ? 90 : -90, scale: .985 }), center: { opacity: 1, x: 0, scale: 1 }, exit: (d: number) => ({ opacity: 0, x: d > 0 ? -70 : 70, scale: .985 }) }} initial="enter" animate="center" exit="exit" transition={{ duration: .42, ease }} className="slide">
              <div className="copy-pane"><div className="kicker">{slide.kicker}</div><h1>{slide.title}</h1><p className="subtitle">{slide.subtitle}</p>{slide.definition && <div className="definition-box"><span>Definición</span><p>{slide.definition}</p></div>}<div className="theory-box"><span>Concepto para estudiar</span><p>{slide.theory}</p></div></div>
              <div className="visual-pane"><Visual type={slide.type} /></div>
            </motion.article>
          )}
        </AnimatePresence>
      </section>

      <footer className="controls"><button onClick={prev} disabled={index === 0}><ArrowLeft size={18} />Anterior</button><div className="dots">{slides.map((_, i) => <button key={i} onClick={() => go(i)} className={i === index ? 'active' : ''} />)}</div><button onClick={next} disabled={index === slides.length - 1}>Siguiente<ArrowRight size={18} /></button></footer>
    </main>
  )
}
